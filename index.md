async def gh_get(path: str):
    url = f"{GITHUB_API}/repos/{GITHUB_REPO}/contents/{path}"
    async with httpx.AsyncClient() as client:
        r = await client.get(url, headers=GH_HEADERS, params={"ref": GITHUB_BRANCH})
    if r.status_code == 404:
        return None, None
    r.raise_for_status()
    data = r.json()
    return base64.b64decode(data["content"]).decode(), data["sha"]

async def gh_put(path: str, content: str, message: str, sha: str | None = None):
    url  = f"{GITHUB_API}/repos/{GITHUB_REPO}/contents/{path}"
    body = {
        "message": message,
        "content": base64.b64encode(content.encode()).decode(),
        "branch":  GITHUB_BRANCH,
    }
    if sha:
        body["sha"] = sha
    async with httpx.AsyncClient() as client:
        r = await client.put(url, headers=GH_HEADERS, json=body)
    r.raise_for_status()
    return r.json()

def parse_ndjson(text: str) -> list[dict]:
    rows = []
    for line in (text or "").splitlines():
        line = line.strip()
        if line:
            try:
                rows.append(json.loads(line))
            except json.JSONDecodeError:
                pass
    return rows

    lines   = (existing or "").rstrip("\n")
    updated = (lines + "\n" + json.dumps(record, ensure_ascii=False)).lstrip("\n")
    await gh_put(path, updated, commit_msg, sha)

@app.get("/slowapi/claims")
async def get_claims(
    resolved: bool = Query(None),
    claims=Depends(verify_token),
):
    """Return claims ledger. Front-end uses this to display prediction accuracy."""
    content, _ = await gh_get(CLAIMS_PATH)
    events = parse_ndjson(content)
    if resolved is not None:
        events = [e for e in events if e.get("resolved") == resolved]
    # compute running mean |ε_FGT| for resolved claims
    resolved_epsilons = [
        abs(e["epsilon_fgt"]) for e in events
        if e.get("event_type") == "resolution" and e.get("epsilon_fgt") is not None
    ]
    mean_fgt = sum(resolved_epsilons) / len(resolved_epsilons) if resolved_epsilons else None
    return {"events": events, "mean_epsilon_fgt": mean_fgt, "slowapi_active": bool(ANTHROPIC_API_KEY or OPENAI_API_KEY)}

                "evidence":   [],

Given this event ledger context, predict the most likely next state of the system.
Return JSON only: {{claim, confidence (0-1), target_ts (ISO), evidence (list of event_ids)}}.

Context:
{json.dumps(context, indent=2)}

    if ANTHROPIC_API_KEY:
        async with httpx.AsyncClient() as client:

                    "messages": [{"role": "user", "content": prompt}],
                },
                timeout=30,

    try:
        prediction = json.loads(raw)
    except Exception:
        prediction = {"claim": raw, "confidence": 0.5, "evidence": []}

    return {"slowapi_active": True, "prediction": prediction}

// ── ROYGBIV color scale ───────────────────────────────────────────────────────
// Maps β ∈ [-1, 1] → ROYGBIV
// R (red)    = strong positive β  (high loss direction, worst)
// V (violet) = strong negative β  (low loss direction, descent / best)
// Green      = near-zero (neutral)
function roygbivColor(beta) {
  // Positive β: R → O → Y → G (red=worst → neutral)
  // Negative β: G → C → B → I → V (neutral → violet=best)
  const v = Math.max(-1, Math.min(1, beta))
  if (v >= 0) {
    // 0 → 1: green → yellow → orange → red
    const t = v
    if (t < 0.33) {
      // green → yellow
      const s = t / 0.33
      return interpolateRGB([74, 222, 128], [250, 204, 21], s)
    } else if (t < 0.66) {
      // yellow → orange
      const s = (t - 0.33) / 0.33
      return interpolateRGB([250, 204, 21], [251, 146, 60], s)
    } else {
      // orange → red
      const s = (t - 0.66) / 0.34
      return interpolateRGB([251, 146, 60], [248, 113, 113], s)
    }
  } else {
    // 0 → -1: green → cyan → blue → indigo → violet
    const t = -v
    if (t < 0.25) {
      const s = t / 0.25
      return interpolateRGB([74, 222, 128], [34, 211, 238], s)
    } else if (t < 0.5) {
      const s = (t - 0.25) / 0.25
      return interpolateRGB([34, 211, 238], [96, 165, 250], s)
    } else if (t < 0.75) {
      const s = (t - 0.5) / 0.25
      return interpolateRGB([96, 165, 250], [129, 140, 248], s)
    } else {
      const s = (t - 0.75) / 0.25
      return interpolateRGB([129, 140, 248], [192, 132, 252], s)
    }
  }
}

function interpolateRGB(a, b, t) {
  const r = Math.round(a[0] + (b[0] - a[0]) * t)
  const g = Math.round(a[1] + (b[1] - a[1]) * t)
  const bl = Math.round(a[2] + (b[2] - a[2]) * t)
  return `rgb(${r},${g},${bl})`
}

function roygbivAlpha(beta) {
  const abs = Math.abs(beta)
  const alpha = 0.18 + abs * 0.82
  return alpha.toFixed(2)
}

function roygbivStyle(beta) {
  const color = roygbivColor(beta)
  const alpha = parseFloat(roygbivAlpha(beta))
  // Extract rgb and apply alpha
  const rgb = color.replace('rgb(', '').replace(')', '').split(',').map(Number)
  return `rgba(${rgb[0]},${rgb[1]},${rgb[2]},${alpha})`
}

  const agent  = LABOR_NAMES[Math.floor(Math.random() * Math.min(staffCount, LABOR_NAMES.length))]
  const [name, role] = agent.split('·')
  const evType = Math.random() < journalRate ? pick(LABOR_EV) : 'sense'
  const noise  = rand(-noiseSigma, noiseSigma)
  let payload
  if (evType === 'journal')
    payload = `intent="${pick(['optimize_throughput','review_FX','compliance_audit','talent_pipeline'])}" ε=${(0.03+noise).toFixed(3)}`
  else if (evType === 'compliance_check')
    payload = `risk=${(0.2+noise).toFixed(2)} status=${Math.random()>0.15?'pass':'⚑flag'}`
  else if (evType === 'regression_target')
    payload = `target=${pick(['equity','fx','derivatives'])} var=${(0.012+noise).toFixed(3)}`
  else
    payload = `ω_log role=${role}`
  return {
    t: nowStr(), agentClass:'labor', agentId:`L_${name.split(' ')[0].toLowerCase()}`,
    agentName:name, role, evType, layer:['journal','token_issued'].includes(evType)?'L2':'L3',
    payload, fleetDelta:0, fxDelta:0,
  }
}

function machineEngine({ fleetSize, sensorHz, faultRate }, st) {
  const m = MACHINE_IDS[Math.floor(Math.random() * Math.min(fleetSize, MACHINE_IDS.length))]
  const [id, type] = m.split('·')
  const isFault = Math.random() < faultRate
  const evType  = isFault ? 'fault_detected' : pick(MACHINE_EV)
  let payload, layer = 'L1'
  if (type.startsWith('HGV') || type.startsWith('Van')) {
    const fuel = rand(80,180).toFixed(1), spd = rand(0,95*sensorHz/5).toFixed(0)
    payload = `fuel=${fuel}L spd=${spd}km/h${isFault?' ⚑FAULT':''}`
  } else {
    const cpu = rand(12,78).toFixed(0), lat = rand(1.1,2.9).toFixed(1)
    payload = `cpu=${cpu}% lat=${lat}ms${isFault?' ⚑OVERLOAD':''}`
    layer = 'L2'
  }
  return {
    t: nowStr(), agentClass:'machine', agentId:id, agentName:type, role:'sensor',
    evType, layer, payload, fleetDelta:isFault?-0.4:0.05, fxDelta:0,
  }
}

function marketEngine({ fxBase, fxVolatility, bondYield, equityDrift }, st) {
  const m = MARKET_IDS[Math.floor(Math.random() * MARKET_IDS.length)]
  const [id, name] = m.split('·')
  let evType = 'market_tick', payload, fxDelta = 0
  if (id === 'FX_UGXUSD') {
    fxDelta = rand(-fxVolatility, fxVolatility)
    const rate = clamp(st.fx + fxDelta, fxBase*0.85, fxBase*1.15)
    payload = `USD/UGX=${rate.toFixed(0)} Δ${fxDelta>0?'+':''}${fxDelta.toFixed(0)}`
  } else if (id === 'BOND_UG10Y') {
    const bDelta = rand(-0.08,0.08)
    payload = `yield=${(bondYield+bDelta).toFixed(2)}% Δ${bDelta>0?'+':''}${(bDelta*100).toFixed(0)}bps`
    if (Math.abs(bDelta) > 0.06) evType = 'shock'
  } else if (id === 'EQ_USE') {
    const eqDelta = rand(equityDrift-0.005, equityDrift+0.005)
    payload = `USE=${(100+eqDelta*100).toFixed(2)} drift=${eqDelta>0?'+':''}${(eqDelta*100).toFixed(2)}%`
  } else {
    payload = `${name} tick=${rand(0.95,1.05).toFixed(4)}`
  }
  return {
    t: nowStr(), agentClass:'market', agentId:id, agentName:name, role:'ticker',
    evType, layer:'L2', payload, fleetDelta:0, fxDelta,
  }
}

// ── Initial Jacobian ──────────────────────────────────────────────────────────
function initJacobian() {
  const driven = {0:[0,1],1:[0,2],2:[1,3],3:[2],4:[3,4],5:[0,1],6:[1,3],7:[2,3],8:[4],9:[4]}
  return Array.from({length:5},(_,oi) =>
    Array.from({length:10},(_,pi) => {
      const isDriven = (driven[pi]||[]).includes(oi)
      const base = isDriven ? rand(0.40,0.88) : rand(0.02,0.22)
      return base * (Math.random()>0.5?1:-1)
    })
  )
}
